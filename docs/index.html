class Human:
    def __init__(self, name, years):
        self.name = name
        self.years = years
    def getInfo(self):
        print(f"Меня зовут {self.name}, мне {self.years} лет.")
class Ivan(Human):
    def __init__(self, name, years, food):
        super.__init__(name, years)
        self.food = food
    def getInfo(self):
        super.getInfo()
        print(f"Я люблю {self.food}!")
ivan = Ivan("Иван", "14", "бургеры")
ivan.getInfo()
